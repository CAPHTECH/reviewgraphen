000056  fn materialize_literate(path: &Path) -> Result<Option<LiterateState>, String> {
000057      if path.extension().and_then(std::ffi::OsStr::to_str) != Some("md") {
000058          return Ok(None);
000059      }
000060      let raw =
000061          std::fs::read_to_string(path).map_err(|error| format!("{}: {error}", path.display()))?;
000062      let blanked = fsl_syntax::extract_literate_fsl(&raw).ok_or_else(|| {
000063          format!(
000064              "{}: Markdown file does not contain any ```fsl fenced code blocks",
000065              path.display()
000066          )
000067      })?;
000068      let stem = path
000069          .file_stem()
000070          .and_then(std::ffi::OsStr::to_str)
000071          .unwrap_or("literate");
000072      // Each CLI process owns its materialization. The original Markdown path is
000073      // passed separately as the stable verify-cache identity, so physical
000074      // isolation does not trade away cache hits across invocations.
000075      let materialized = literate_materialization_path(path, stem, std::process::id());
000076      std::fs::write(&materialized, &blanked).map_err(|error| error.to_string())?;
000077      Ok(Some(LiterateState { path: materialized }))
000078  }
000079  
000080  fn literate_materialization_path(path: &Path, stem: &str, process_id: u32) -> PathBuf {
000081      path.with_file_name(format!(".{stem}.literate-{process_id}.fsl"))
000082  }
