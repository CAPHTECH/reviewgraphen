000052  fn materialize_literate(path: &Path) -> Result<Option<LiterateState>, String> {
000053      if path.extension().and_then(std::ffi::OsStr::to_str) != Some("md") {
000054          return Ok(None);
000055      }
000056      let raw =
000057          std::fs::read_to_string(path).map_err(|error| format!("{}: {error}", path.display()))?;
000058      let blanked = fsl_syntax::extract_literate_fsl(&raw).ok_or_else(|| {
000059          format!(
000060              "{}: Markdown file does not contain any ```fsl fenced code blocks",
000061              path.display()
000062          )
000063      })?;
000064      let stem = path
000065          .file_stem()
000066          .and_then(std::ffi::OsStr::to_str)
000067          .unwrap_or("literate");
000068      // Each CLI process owns its materialization. The original Markdown path is
000069      // passed separately as the stable verify-cache identity, so physical
000070      // isolation does not trade away cache hits across invocations.
000071      let materialized = literate_materialization_path(path, stem, std::process::id());
000072      std::fs::write(&materialized, &blanked).map_err(|error| error.to_string())?;
000073      Ok(Some(LiterateState { path: materialized }))
000074  }
000075  
000076  fn literate_materialization_path(path: &Path, stem: &str, process_id: u32) -> PathBuf {
000077      path.with_file_name(format!(".{stem}.literate-{process_id}.fsl"))
000078  }
