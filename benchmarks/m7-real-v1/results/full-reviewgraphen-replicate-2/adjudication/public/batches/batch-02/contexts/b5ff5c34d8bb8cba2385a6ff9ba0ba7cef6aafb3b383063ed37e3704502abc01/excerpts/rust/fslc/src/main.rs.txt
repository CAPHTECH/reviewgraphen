000055  fn materialize_literate(path: &Path) -> Result<Option<LiterateState>, String> {
000056      if path.extension().and_then(std::ffi::OsStr::to_str) != Some("md") {
000057          return Ok(None);
000058      }
000059      let raw =
000060          std::fs::read_to_string(path).map_err(|error| format!("{}: {error}", path.display()))?;
000061      let blanked = fsl_syntax::extract_literate_fsl(&raw).ok_or_else(|| {
000062          format!(
000063              "{}: Markdown file does not contain any ```fsl fenced code blocks",
000064              path.display()
000065          )
000066      })?;
000067      let stem = path
000068          .file_stem()
000069          .and_then(std::ffi::OsStr::to_str)
000070          .unwrap_or("literate");
000071      // Each CLI process owns its materialization. The original Markdown path is
000072      // passed separately as the stable verify-cache identity, so physical
000073      // isolation does not trade away cache hits across invocations.
000074      let materialized = literate_materialization_path(path, stem, std::process::id());
000075      std::fs::write(&materialized, &blanked).map_err(|error| error.to_string())?;
000076      Ok(Some(LiterateState { path: materialized }))
000077  }
000078  
000079  fn literate_materialization_path(path: &Path, stem: &str, process_id: u32) -> PathBuf {
000080      path.with_file_name(format!(".{stem}.literate-{process_id}.fsl"))
000081  }
