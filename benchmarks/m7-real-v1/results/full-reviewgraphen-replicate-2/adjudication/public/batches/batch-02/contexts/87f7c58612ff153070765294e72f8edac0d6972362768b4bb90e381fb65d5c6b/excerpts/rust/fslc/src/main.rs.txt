000043  fn materialize_literate(path: &Path) -> Result<Option<LiterateState>, String> {
000044      if path.extension().and_then(std::ffi::OsStr::to_str) != Some("md") {
000045          return Ok(None);
000046      }
000047      let raw =
000048          std::fs::read_to_string(path).map_err(|error| format!("{}: {error}", path.display()))?;
000049      let blanked = fsl_syntax::extract_literate_fsl(&raw).ok_or_else(|| {
000050          format!(
000051              "{}: Markdown file does not contain any ```fsl fenced code blocks",
000052              path.display()
000053          )
000054      })?;
000055      let stem = path
000056          .file_stem()
000057          .and_then(std::ffi::OsStr::to_str)
000058          .unwrap_or("literate");
000059      // Each CLI process owns its materialization. The original Markdown path is
000060      // passed separately as the stable verify-cache identity, so physical
000061      // isolation does not trade away cache hits across invocations.
000062      let materialized = literate_materialization_path(path, stem, std::process::id());
000063      std::fs::write(&materialized, &blanked).map_err(|error| error.to_string())?;
000064      Ok(Some(LiterateState { path: materialized }))
000065  }
000066  
000067  fn literate_materialization_path(path: &Path, stem: &str, process_id: u32) -> PathBuf {
000068      path.with_file_name(format!(".{stem}.literate-{process_id}.fsl"))
000069  }
