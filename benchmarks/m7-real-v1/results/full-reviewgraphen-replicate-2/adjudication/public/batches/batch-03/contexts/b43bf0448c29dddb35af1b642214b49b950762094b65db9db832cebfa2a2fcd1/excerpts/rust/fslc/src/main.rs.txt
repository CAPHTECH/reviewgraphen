000320  fn main() {
000321      // `run` on an explicitly sized thread rather than the process's main
000322      // thread, whose size the OS picks.
000323      let worker = std::thread::Builder::new()
000324          .stack_size(STACK_SIZE)
000325          .name("fslc".to_owned())
000326          .spawn(run)
000327          .expect("spawn the fslc worker thread");

000329          // The panic message has already been printed by the default hook.

000332          std::process::exit(101);

000337      if print_cli_metadata() {
000338          return;
000339      }
000340      if let Some((output, status)) = fmt_command() {
000341          match output {
000342              FmtCliOutput::Source(source) => print!("{source}"),
000343              FmtCliOutput::Json(value) => println!(
000344                  "{}",
000345                  serde_json::to_string_pretty(&value).expect("serialize fmt output")
000346              ),
000347          }
000348          if status != 0 {
000349              std::process::exit(status);
000350          }
000351          return;
000352      }
000353      let (output, reported_status) = match command() {
000354          Ok(result) => result,
000355          Err(error) => (error_output("usage", &error), 2),
000356      };
000357      let status = normalized_exit_status(&output, reported_status);
000358      println!(
000359          "{}",
000360          serde_json::to_string_pretty(&output).expect("serialize CLI output")
000361      );
000362      if status != 0 {
000363          std::process::exit(status);
000364      }
000365  }
