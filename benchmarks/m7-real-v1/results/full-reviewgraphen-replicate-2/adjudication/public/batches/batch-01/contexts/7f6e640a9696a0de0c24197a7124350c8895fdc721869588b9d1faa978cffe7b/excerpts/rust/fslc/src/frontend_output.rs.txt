000104  pub fn is_ai_project(source: &str) -> bool {
000105      const PROJECT_BLOCKS: &[&str] = &[
000106          "ai_action",
000107          "ai_component",
000108          "ai_contract",
000109          "ai_migration",
000110          "authority",
000111          "dataset",
000112          "evaluator",
000113          "failure_mode",
000114          "observed_property",
000115          "retriever",
000116          "statistical_property",
000117          "trust_boundary",
000118      ];
000119      let has_project_property = source.lines().any(|line| {
000120          let line = line.trim_start();
000121          line.starts_with("statistical_property ")
000122              || line.starts_with("ai_migration ")
000123              || line.starts_with("observed_property ")
000124      });
000125      has_project_property
000126          && fsl_syntax::declaration_keyword(source)
000127              .is_ok_and(|keyword| PROJECT_BLOCKS.contains(&keyword.as_str()))
000128  }
