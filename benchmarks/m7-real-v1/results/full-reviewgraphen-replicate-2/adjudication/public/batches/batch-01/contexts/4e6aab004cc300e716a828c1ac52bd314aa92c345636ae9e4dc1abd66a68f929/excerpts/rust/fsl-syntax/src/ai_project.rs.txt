000294  /// Parse one fsl-ai project source into its typed declarations.
000295  ///
000296  /// # Errors
000297  ///
000298  /// Returns a message when no recognized top-level block is found, a block is
000299  /// unterminated, or a `statistical_property`/`observed_property`/
000300  /// `ai_migration` name is declared more than once.
000301  pub fn parse_ai_project(source: &str, name: &str) -> Result<AiProject, String> {
000302      let blocks = top_blocks(source)?;
000303      if blocks.is_empty() {
000304          return Err("expected fsl-ai project declarations".to_owned());
000305      }
000306      let mut project = AiProject {
000307          name: name.to_owned(),
000308          ..AiProject::default()
000309      };
000310      for block in &blocks {
000311          match block.kind.as_str() {
000312              "ai_component" => {
000313                  project
000314                      .components
000315                      .push(parse_ai_component(&block.text).map_err(|error| error.to_string())?);
000316              }
000317              "dataset" => project.datasets.push(parse_dataset(block)),
000318              "statistical_property" => project
000319                  .statistical_properties
000320                  .push(parse_statistical_property(block)?),
000321              "observed_property" => project
000322                  .observed_properties
000323                  .push(parse_observed_property(block)?),
000324              "ai_migration" => project.migrations.push(parse_migration(block)?),
000325              kind if RAW_BLOCKS.contains(&kind) => project.raw_blocks.push(AiRawBlock {
000326                  kind: block.kind.clone(),
000327                  name: block.name.clone(),
000328              }),
000329              _ => {}
000330          }
000331      }
000332      reject_duplicates(
000333          project
000334              .statistical_properties
000335              .iter()
000336              .map(|p| p.name.as_str()),
000337          "statistical_property",
000338      )?;
000339      reject_duplicates(
000340          project.observed_properties.iter().map(|p| p.name.as_str()),
000341          "observed_property",
000342      )?;
000343      reject_duplicates(
000344          project.migrations.iter().map(|p| p.name.as_str()),
000345          "ai_migration",
000346      )?;
000347      Ok(project)
000348  }
