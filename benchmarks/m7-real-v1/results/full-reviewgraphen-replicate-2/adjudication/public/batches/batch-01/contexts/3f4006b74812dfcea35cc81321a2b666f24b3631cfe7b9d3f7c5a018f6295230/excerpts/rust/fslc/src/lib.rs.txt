000096  pub fn conformance_vectors_for_version(
000097      model: &KernelModel,
000098      depth: usize,
000099      version: fsl_core::PublicKernelVersion,
000100  ) -> Result<Value, String> {
000101      let all_calls = action_calls(model)?;
000102      let initial = fsl_runtime::Monitor::new(model.clone()).map_err(|error| error.to_string())?;
000103      let initial_json = conformance_state_json(model, &initial.state)?;
000104      let initial_key = serde_json::to_string(&initial_json).map_err(|error| error.to_string())?;
000105      let mut seen = BTreeMap::from([(initial_key, "s0".to_owned())]);
000106      let mut queue = VecDeque::from([("s0".to_owned(), 0_usize, initial)]);
000107      let mut states = vec![json!({"id":"s0","depth":0,"state":initial_json})];
000108      let mut vectors = Vec::new();
000109  
000110      while let Some((state_id, state_depth, monitor)) = queue.pop_front() {
000111          let before = conformance_state_json(model, &monitor.state)?;
000112          for (action, params) in &all_calls {
000113              let action_json = json!({
000114                  "name":display_name(action),
000115                  "params":params.iter().map(|(name,value)|(name.clone(),fsl_value_json(value))).collect::<serde_json::Map<_,_>>()
000116              });
000117              let mut successor = monitor.clone();
000118              let result = successor
000119                  .attempt(action, params)
000120                  .map_err(|error| error.to_string())?;
000121              let after = conformance_state_json(model, &result.state)?;
000122              if let Some(violation) = result.violation {
000123                  let attempted = result
000124                      .attempted_state
000125                      .as_ref()
000126                      .map(|state| conformance_state_json(model, state))
000127                      .transpose()?;
000128                  vectors.push(json!({
000129                      "state":state_id,"action":action_json,
000130                      "outcome":{
000131                          "kind":violation.kind,"name":violation.name,
000132                          "state_changed":after != before,"state":after,
000133                          "attempted_state":attempted
000134                      }
000135                  }));
000136                  continue;
000137              }
000138  
000139              let changes = compute_changes(&before, &after);
000140              vectors.push(json!({
000141                  "state":state_id,"action":action_json,
000142                  "outcome":{"kind":"ok","state_changed":after != before,"state":after,"changes":changes}
000143              }));
000144              if state_depth >= depth {
000145                  continue;
000146              }
000147              let key = serde_json::to_string(&after).map_err(|error| error.to_string())?;
000148              if let std::collections::btree_map::Entry::Vacant(entry) = seen.entry(key) {
000149                  let id = format!("s{}", states.len());
000150                  entry.insert(id.clone());
000151                  states.push(json!({"id":id,"depth":state_depth+1,"state":after}));
000152                  queue.push_back((id, state_depth + 1, successor));
000153              }
000154          }
000155      }
000156  
000157      let (schema_id, schema_version) = match version {
000158          fsl_core::PublicKernelVersion::V1 => {
000159              (CONFORMANCE_V1_SCHEMA_ID, CONFORMANCE_V1_SCHEMA_VERSION)
000160          }
000161          fsl_core::PublicKernelVersion::V2 => {
000162              (CONFORMANCE_V2_SCHEMA_ID, CONFORMANCE_V2_SCHEMA_VERSION)
000163          }
000164      };
000165      Ok(json!({
000166          "$schema":schema_id,
000167          "schema_version":schema_version,
000168          "kernel_schema_version":version.schema_version(),
000169          "result":"conformance",
000170          "spec":model.name,
000171          "depth":depth,
000172          "states":states,
000173          "vectors":vectors,
000174      }))
000175  }

000424  fn action_calls(model: &KernelModel) -> Result<Vec<ActionCall>, String> {
000425      let mut calls = Vec::new();
000426      for action in &model.actions {
000427          let mut bindings = vec![BTreeMap::new()];
000428          for parameter in &action.params {
000429              let values = match parameter {
000430                  ParamDef::Typed { ty, .. } => {
000431                      model.domain_values(ty).map_err(|error| error.to_string())?
000432                  }
000433                  ParamDef::Range { lo, hi, .. } => (*lo..=*hi).map(FslValue::Int).collect(),
000434              };
000435              let mut next = Vec::new();
000436              for existing in bindings {
000437                  for value in &values {
000438                      let mut candidate = existing.clone();
000439                      candidate.insert(parameter.name().to_owned(), value.clone());
000440                      next.push(candidate);
000441                  }
000442              }
000443              bindings = next;
000444          }
000445          calls.extend(
000446              bindings
000447                  .into_iter()
000448                  .map(|params| (action.name.clone(), params)),
000449          );
000450      }
000451      calls.sort_by(|left, right| {
000452          left.0
000453              .cmp(&right.0)
000454              .then_with(|| format!("{:?}", left.1).cmp(&format!("{:?}", right.1)))
000455      });
000456      let mut unique = BTreeSet::new();
000457      calls.retain(|call| unique.insert(format!("{}:{:?}", call.0, call.1)));
000458      Ok(calls)
