000388  fn try_match_header(chars: &[char], start: usize) -> Option<BlockHeader> {
000389      let n = chars.len();
000390      let mut p = start;
000391      while p < n && is_ident_continue(chars[p]) {
000392          p += 1;
000393      }
000394      let kind: String = chars[start..p].iter().collect();
000395      let mut q = p;
000396      while q < n && chars[q].is_whitespace() {
000397          q += 1;
000398      }
000399      if q > p && q < n && is_ident_start(chars[q]) {
000400          let name_start = q;
000401          let mut r = q;
000402          while r < n && is_ident_continue(chars[r]) {
000403              r += 1;
000404          }
000405          let mut s = r;
000406          while s < n && chars[s].is_whitespace() {
000407              s += 1;
000408          }
000409          if s < n && chars[s] == '{' {
000410              return Some(BlockHeader {
000411                  start,
000412                  kind,
000413                  name: chars[name_start..r].iter().collect(),
000414                  brace: s,
000415              });
000416          }
000417      }
000418      let mut s = p;
000419      while s < n && chars[s].is_whitespace() {
000420          s += 1;
000421      }
000422      if s < n && chars[s] == '{' {
000423          return Some(BlockHeader {
000424              start,
000425              kind,
000426              name: String::new(),
000427              brace: s,
000428          });
000429      }
000430      None

000433  fn find_block_header(chars: &[char], from: usize) -> Option<BlockHeader> {
000434      let n = chars.len();
000435      let mut pos = from;
000436      while pos < n {
000437          if is_ident_start(chars[pos])
000438              && (pos == 0 || !is_ident_continue(chars[pos - 1]))
000439              && let Some(header) = try_match_header(chars, pos)
000440          {
000441              return Some(header);
000442          }
000443          pos += 1;
000444      }
000445      None
000446  }
