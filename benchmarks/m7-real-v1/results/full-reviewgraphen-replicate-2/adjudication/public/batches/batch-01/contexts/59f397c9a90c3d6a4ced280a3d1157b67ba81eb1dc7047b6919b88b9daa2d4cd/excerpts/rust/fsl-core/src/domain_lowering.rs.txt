000098  fn lower_name(name: &str) -> String {
000099      let mut output = String::new();
000100      let characters = name.chars().collect::<Vec<_>>();
000101      for (index, character) in characters.iter().enumerate() {
000102          let previous = index.checked_sub(1).and_then(|index| characters.get(index));
000103          let next = characters.get(index + 1);
000104          if character.is_ascii_uppercase()
000105              && index > 0
000106              && (previous.is_some_and(char::is_ascii_lowercase)
000107                  || previous.is_some_and(char::is_ascii_digit)
000108                  || next.is_some_and(char::is_ascii_lowercase))
000109          {
000110              output.push('_');
000111          }
000112          output.push(character.to_ascii_lowercase());
000113      }
000114      safe(&output)
000115  }

000168          "TimedOut"

000170          "Failed"

000172          "Cancelled"
000173      } else {
000174          "Succeeded"

000179      for effect in &domain.effects {
000180          if let Some(message) = effect.outcome_role_conflict() {
000181              return Err(error_at(message, span_at(effect.loc)));
000182          }
000183          for event in effect.explicit_outcome_events() {
000184              if !domain
000185                  .aggregates
000186                  .iter()
000187                  .flat_map(|aggregate| &aggregate.events)
000188                  .any(|candidate| candidate.name == *event)
000189              {
000190                  return Err(error_at(
000191                      format!("unknown domain event '{event}'"),
000192                      span_at(effect.loc),

000194              }
000195          }
000196      }
