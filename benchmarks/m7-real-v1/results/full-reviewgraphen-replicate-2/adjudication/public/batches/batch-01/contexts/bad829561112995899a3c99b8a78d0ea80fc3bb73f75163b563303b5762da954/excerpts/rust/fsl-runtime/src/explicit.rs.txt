000401  fn collect_assigned_init_roots(statements: &[Statement], assigned: &mut BTreeSet<String>) {
000402      for statement in statements {
000403          match statement {
000404              Statement::Assign { target, .. } => {
000405                  if let Some(name) = logical_var(target) {
000406                      assigned.insert(name.to_owned());
000407                  }
000408              }
000409              Statement::If {
000410                  then_statements,
000411                  else_statements,
000412                  ..
000413              } => {
000414                  collect_assigned_init_roots(then_statements, assigned);
000415                  collect_assigned_init_roots(else_statements, assigned);
000416              }
000417              Statement::ForAll { statements, .. } => {
000418                  collect_assigned_init_roots(statements, assigned);
000419              }
000420          }
000421      }
000422  }
