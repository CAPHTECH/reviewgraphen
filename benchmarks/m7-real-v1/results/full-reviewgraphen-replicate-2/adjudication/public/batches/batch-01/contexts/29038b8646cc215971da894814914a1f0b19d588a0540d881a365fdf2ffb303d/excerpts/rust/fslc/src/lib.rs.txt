000100  ///
000101  /// # Errors
000102  ///
000103  /// Returns an error when concrete initialization or exploration fails.
000104  pub fn conformance_vectors_for_version(
000105      model: &KernelModel,
000106      depth: usize,
000107      version: fsl_core::PublicKernelVersion,
000108  ) -> Result<Value, String> {
000109      let all_calls = action_calls(model)?;
000110      let initial = fsl_runtime::Monitor::new(model.clone()).map_err(|error| error.to_string())?;
000111      let initial_json = conformance_state_json(model, &initial.state)?;
000112      let initial_key = serde_json::to_string(&initial_json).map_err(|error| error.to_string())?;
000113      let mut seen = BTreeMap::from([(initial_key, "s0".to_owned())]);
000114      let mut queue = VecDeque::from([("s0".to_owned(), 0_usize, initial)]);
000115      let mut states = vec![json!({"id":"s0","depth":0,"state":initial_json})];
000116      let mut vectors = Vec::new();
000117  
000118      while let Some((state_id, state_depth, monitor)) = queue.pop_front() {
000119          let before = conformance_state_json(model, &monitor.state)?;
000120          for (action, params) in &all_calls {
000121              let action_json = json!({
000122                  "name":display_name(action),
000123                  "params":params.iter().map(|(name,value)|(name.clone(),fsl_value_json(value))).collect::<serde_json::Map<_,_>>()
000124              });
000125              let mut successor = monitor.clone();
000126              let result = successor
000127                  .attempt(action, params)
000128                  .map_err(|error| error.to_string())?;
000129              let after = conformance_state_json(model, &result.state)?;
000130              if let Some(violation) = result.violation {
000131                  let attempted = result
000132                      .attempted_state
000133                      .as_ref()
000134                      .map(|state| conformance_state_json(model, state))
000135                      .transpose()?;
000136                  vectors.push(json!({
000137                      "state":state_id,"action":action_json,
000138                      "outcome":{
000139                          "kind":violation.kind,"name":violation.name,
000140                          "state_changed":after != before,"state":after,
000141                          "attempted_state":attempted
000142                      }
000143                  }));
000144                  continue;
000145              }
000146  
000147              let changes = compute_changes(&before, &after);
000148              vectors.push(json!({
000149                  "state":state_id,"action":action_json,
000150                  "outcome":{"kind":"ok","state_changed":after != before,"state":after,"changes":changes}
000151              }));
000152              if state_depth >= depth {
000153                  continue;
000154              }
000155              let key = serde_json::to_string(&after).map_err(|error| error.to_string())?;
000156              if let std::collections::btree_map::Entry::Vacant(entry) = seen.entry(key) {
000157                  let id = format!("s{}", states.len());
000158                  entry.insert(id.clone());
000159                  states.push(json!({"id":id,"depth":state_depth+1,"state":after}));
000160                  queue.push_back((id, state_depth + 1, successor));
000161              }
000162          }
000163      }
000164  
000165      let (schema_id, schema_version) = match version {
000166          fsl_core::PublicKernelVersion::V1 => {

000168          }
000169          fsl_core::PublicKernelVersion::V2 => {
000170              (CONFORMANCE_V2_SCHEMA_ID, CONFORMANCE_V2_SCHEMA_VERSION)
000171          }
000172      };
000173      Ok(json!({
000174          "$schema":schema_id,
000175          "schema_version":schema_version,
000176          "kernel_schema_version":version.schema_version(),
000177          "result":"conformance",
000178          "spec":model.name,
000179          "depth":depth,
000180          "states":states,
000181          "vectors":vectors,
000182      }))
