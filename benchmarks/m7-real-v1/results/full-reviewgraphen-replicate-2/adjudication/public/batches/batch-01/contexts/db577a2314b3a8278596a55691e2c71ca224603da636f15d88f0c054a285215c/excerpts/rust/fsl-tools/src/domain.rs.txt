000257  /// Generate target files from Public Kernel v1 plus the versioned compatibility
000258  /// metadata needed for source-level DDD names that lowering does not retain.
000259  ///
000260  /// # Errors
000261  ///
000262  /// Returns an error when either input contract is incompatible or inconsistent,
000263  /// or when the target is unsupported.
000264  pub fn domain_scaffold(kernel: &Value, metadata: &Value, target: &str) -> Result<Value, String> {
000265      let files = crate::domain_codegen::generate(kernel, metadata, target)?;
000266      let domain = kernel
000267          .pointer("/spec/name")
000268          .and_then(Value::as_str)
000269          .ok_or_else(|| "public Kernel root.spec.name must be a string".to_owned())?;
000270      Ok(json!({
000271          "result":"generated",
000272          "dialect":"fsl-domain-effect.v0",
000273          "domain":domain,
000274          "target":target,
000275          "files":files.into_iter().map(|(path, content)| json!({"path":path,"content":content})).collect::<Vec<_>>()
000276      }))
000277  }

000279  /// Reuse the TypeScript generator for the adapter snippets embedded by
000280  /// `domain testgen`; this prevents a second adapter/effect implementation.
000281  ///
000282  /// # Errors
000283  ///
000284  /// Returns an error when the Public Kernel or metadata contract is invalid or
000285  /// inconsistent.
000286  pub fn domain_adapter_files(
000287      kernel: &Value,
000288      metadata: &Value,
000289  ) -> Result<Vec<(String, String)>, String> {
000290      let mut files = crate::domain_codegen::generate(kernel, metadata, "typescript")?
000291          .into_iter()
000292          .filter(|(path, _)| path.ends_with("/adapter.ts") || path == "effects.ts")
000293          .collect::<Vec<_>>();
000294      files.sort_by_key(|(path, _)| (path == "effects.ts", path.clone()));
000295      Ok(files)
