000318  /// Parse a direct spec/business/requirements document with temporary verify-bound overrides.
000319  ///
000320  /// This is the semantic seam used by the `sweep` driver; it does not mutate
000321  /// source files or the ordinary parser result.
000322  ///
000323  /// # Errors
000324  ///
000325  /// Returns [`CoreError`] for parse, lowering, or unsupported top-level input.
000326  pub fn parse_kernel_source_with_bounds(
000327      source: &str,
000328      instances: &std::collections::BTreeMap<String, i64>,
000329      values: &std::collections::BTreeMap<String, (i64, i64)>,
000330  ) -> Result<KernelSpec, CoreError> {
000331      fn update_bounds(
000332          items: &mut [VerifyItem],
000333          instances: &std::collections::BTreeMap<String, i64>,
000334          values: &std::collections::BTreeMap<String, (i64, i64)>,
000335      ) {
000336          for bound in items {
000337              match bound {
000338                  VerifyItem::Instances(name, count, _) => {
000339                      if let Some(value) = instances.get(name) {
000340                          *count = *value;
000341                      }
000342                  }
000343                  VerifyItem::Values(name, lo, hi, _) => {
000344                      if let Some((new_lo, new_hi)) = values.get(name) {
000345                          **lo = Expr::Num(*new_lo);
000346                          **hi = Expr::Num(*new_hi);
000347                      }
000348                  }
000349              }
000350          }
000351      }
000352      fn update(
000353          item: &mut SpecItem,
000354          instances: &std::collections::BTreeMap<String, i64>,
000355          values: &std::collections::BTreeMap<String, (i64, i64)>,
000356      ) {
000357          if let SpecItem::VerifyBounds { items, .. } = item {
000358              update_bounds(items, instances, values);
000359          }
000360      }
000361  
000362      let parsed = parse_document(SourceFile::new(source))?;
000363      let mut kernel = match parsed.surface {
000364          SurfaceDocument::Spec(mut spec) => {
000365              validate_direct_scope_overrides(&spec, instances, values)?;
000366              for item in &mut spec.items {
000367                  update(item, instances, values);
000368              }
000369              lower_direct_spec(spec)
000370          }
000371          SurfaceDocument::Business(mut business) => {
000372              for item in &mut business.items {
000373                  if let BusinessItem::VerifyBounds { items, .. } = item {
000374                      update_bounds(items, instances, values);
000375                  }
000376              }
000377              lower_business(business)
000378          }
000379          SurfaceDocument::Requirements(mut requirements) => {
000380              for item in &mut requirements.items {
000381                  if let RequirementsItem::Common(item) = item {
000382                      update(item, instances, values);
000383                  }
000384              }
000385              lower_requirements(requirements)
000386          }
000387          _ => Err(CoreError {
000388              message: "scope overrides only support spec, business, or requirements".to_owned(),
000389              line: 1,
000390              column: 1,
000391              origin: None,
000392          }),
000393      }?;
000394      kernel.annotations.extend(SPEC_TARGET, parsed.annotations);
000395      Ok(kernel)
000396  }
