000007  /// Render a surface-parser diagnostic using the public check/verify envelope.
000008  #[must_use]
000009  pub fn render_surface_parse_error(
000010      mut output: Map<String, Value>,
000011      error: &fsl_syntax::ParseError,
000012  ) -> Value {
000013      output.insert("result".to_owned(), json!("error"));
000014      output.insert("kind".to_owned(), json!("parse"));
000015      output.insert("message".to_owned(), json!(error.to_string()));
000016      output.insert("diagnostic_code".to_owned(), json!(error.code()));
000017      output.insert("loc".to_owned(), error.span.python_loc());
000018      if matches!(
000019          error.code(),
000020          "FSL-DIALECT-EMPTY" | "FSL-DIALECT-ANNOTATION-TARGET" | "FSL-DIALECT-UNKNOWN"
000021      ) {
000022          output.insert(
000023              "supported_dialects".to_owned(),
000024              json!(fsl_syntax::DIALECT_KEYWORDS),
000025          );
000026      }
000027      Value::Object(output)
000028  }

000052  pub fn parse_checked_ai_project(
000053      source: &str,
000054      name: &str,
000055  ) -> Result<fsl_syntax::AiProject, AiProjectParseError> {
000056      let project =
000057          fsl_syntax::parse_ai_project(source, name).map_err(|message| AiProjectParseError {
000058              message,
000059              position: None,
000060          })?;
000061      let unparsed = project.unparsed_clauses();
000062      if unparsed.is_empty() {
000063          return Ok(project);
000064      }
000065      let detail = unparsed
000066          .iter()
000067          .map(|clause| {
000068              format!(
000069                  "{} '{}' (slice '{}'): require {}",
000070                  clause.declaration_kind, clause.declaration, clause.slice, clause.source
000071              )
000072          })
000073          .collect::<Vec<_>>()
000074          .join("; ");
000075      // `unparsed_clauses` walks declarations in source order, so the first entry
000076      // is the first offending clause; report its own line, not its block's.
000077      let first = &unparsed[0];
000078      Err(AiProjectParseError {
000079          message: format!(
000080              "require clause matches no known fsl-ai evidence clause grammar \
000081               (min_samples, ci_lower, ci_upper, a point estimate, observed, or drift): {detail}"
000082          ),
000083          position: Some((first.line, first.column)),
000084      })
000085  }

000095  /// Render the multi-declaration AI project check result when applicable,
000096  /// paired with its process exit code.
000097  #[must_use]
000098  pub fn ai_project_check_output(
000099      source: &str,
000100      source_file: &str,
000101      mut output: Map<String, Value>,
000102  ) -> Option<(Value, i32)> {
000103      if !is_ai_project(source) {
000104          return None;
000105      }
000106      let spec = ai_project_name(source_file);
000107      if let Err(error) = parse_checked_ai_project(source, spec) {
000108          output.insert("result".to_owned(), json!("error"));
000109          output.insert("kind".to_owned(), json!("parse"));
000110          output.insert("message".to_owned(), json!(error.message));
000111          if let Some((line, column)) = error.position {
000112              output.insert("loc".to_owned(), json!({"line": line, "column": column}));
000113          }
000114          return Some((Value::Object(output), 2));
000115      }
000116      output.insert("result".to_owned(), json!("ok"));
000117      output.insert("spec".to_owned(), json!(spec));
000118      output.insert("dialect".to_owned(), json!("fsl-ai-project.v0"));
000119      output.insert("warnings".to_owned(), json!([]));
000120      output.insert(
000121          "ai_analysis_result".to_owned(),
000122          json!("ai_project_analyzed"),
000123      );
000124      Some((Value::Object(output), 0))
000125  }
