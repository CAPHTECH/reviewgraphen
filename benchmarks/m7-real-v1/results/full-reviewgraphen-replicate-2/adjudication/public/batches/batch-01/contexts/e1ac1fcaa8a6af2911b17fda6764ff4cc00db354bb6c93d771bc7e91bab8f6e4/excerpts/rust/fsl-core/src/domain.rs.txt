000084  fn safe(name: &str) -> String {
000085      let mut value = name
000086          .chars()
000087          .map(|character| {
000088              if character.is_ascii_alphanumeric() || character == '_' {
000089                  character
000090              } else {
000091                  '_'
000092              }
000093          })
000094          .collect::<String>();
000095      if value.is_empty() {
000096          value.push('x');
000097      }
000098      if value.starts_with(|character: char| character.is_ascii_digit()) {
000099          value.insert(0, '_');
000100      }
000101      value
000102  }

000104  fn lower_name(name: &str) -> String {
000105      let mut output = String::new();
000106      let characters = name.chars().collect::<Vec<_>>();
000107      for (index, character) in characters.iter().enumerate() {
000108          let previous = index.checked_sub(1).and_then(|index| characters.get(index));
000109          let next = characters.get(index + 1);
000110          if character.is_ascii_uppercase()
000111              && index > 0
000112              && (previous.is_some_and(char::is_ascii_lowercase)
000113                  || previous.is_some_and(char::is_ascii_digit)
000114                  || next.is_some_and(char::is_ascii_lowercase))
000115          {
000116              output.push('_');
000117          }
000118          output.push(character.to_ascii_lowercase());
000119      }
000120      safe(&output)
000121  }

000123  fn identifiers(type_ref: &str) -> Vec<String> {
000124      let mut output = Vec::new();
000125      let mut current = String::new();
000126      for character in type_ref.chars().chain(std::iter::once(' ')) {
000127          if character.is_ascii_alphanumeric() || character == '_' {
000128              current.push(character);
000129          } else if !current.is_empty() {
000130              if matches!(
000131                  current.as_str(),
000132                  "Int" | "Bool" | "Map" | "Set" | "Seq" | "Option" | "relation"
000133              ) {
000134                  current.clear();
000135              } else {
000136                  output.push(std::mem::take(&mut current));
000137              }
000138          }
000139      }
000140      output
000141  }

000143  fn replace_identifier(source: &str, from: &str, to: &str) -> String {
000144      let mut output = String::new();
000145      let bytes = source.as_bytes();
000146      let mut index = 0;
000147      while index < bytes.len() {
000148          let matches = source[index..].starts_with(from);
000149          let before = index
000150              .checked_sub(1)
000151              .and_then(|position| bytes.get(position))
000152              .is_some_and(|byte| byte.is_ascii_alphanumeric() || *byte == b'_');
000153          let after = bytes
000154              .get(index + from.len())
000155              .is_some_and(|byte| byte.is_ascii_alphanumeric() || *byte == b'_');
000156          if matches && !before && !after {
000157              output.push_str(to);
000158              index += from.len();
000159          } else {
000160              let character = source[index..].chars().next().expect("valid char boundary");
000161              output.push(character);
000162              index += character.len_utf8();
000163          }
000164      }
000165      output
000166  }
